7. Exploratory Data Analysis Techniques

After cleaning, EDA helps you understand distributions, relationships and remaining anomalies. It is iterative: new insights often send you back to cleaning.

Univariate summaries

# Numeric
print(df.describe())

# Categorical
for col in df.select_dtypes(include=["object", "category"]).columns:
    print(f"\n{col}:")
    print(df[col].value_counts(dropna=False).head(10))

Distributions

import matplotlib.pyplot as plt
import seaborn as sns

# Histogram + KDE
sns.histplot(df["salary"], kde=True)
plt.title("Salary distribution")
plt.show()

# Count plot for categories
sns.countplot(y=df["department"], order=df["department"].value_counts().index)
plt.title("Department counts")
plt.show()

Relationships

# Correlation matrix (numeric)
corr = df.select_dtypes(include="number").corr()
sns.heatmap(corr, annot=True, cmap="coolwarm", center=0)
plt.title("Correlation heatmap")
plt.show()

# Scatter with optional hue
sns.scatterplot(data=df, x="age", y="salary", hue="department")
plt.show()

# Pair plot for a few numeric columns
sns.pairplot(df[["age", "salary", "tenure"]])
plt.show()

Grouped summaries

# Mean salary by department
print(df.groupby("department")["salary"].agg(["mean", "median", "count"]))

# Crosstab
print(pd.crosstab(df["department"], df["status"], normalize="index"))
